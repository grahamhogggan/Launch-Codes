#telemetry#
fetch DS distance dist;
compare 1000 dist close;
compare dist 0 valid;
AND close valid valid;
AND valid landing showDist;
printif showDist dist;
fetch VS directional landingVelocity;
printif landing landingVelocity;
#main#

branch coasting COAST;
goto NOTCOAST;
lbl COAST;
set desiredAngle 180;
call StabilizeCSM;
fetch AngleSensor angle csmRot;
compare 190 csmRot plus;
compare csmRot 170 minus;
AND plus minus fireEngine;
branch fireEngine THRUSTCSM;
CsmEngine bind w;
goto ENDCOAST;

lbl THRUSTCSM;
CsmEngine unbind;
CsmEngine startup;
lbl ENDCOAST;
loop;
lbl NOTCOAST;
branch landing LANDING;
loop;

lbl LANDING; 
fetch GPS y altitude;
print altitude;
set desiredAngle 0;
compare altitude 7800 close;
branch close AUTOLAND;
print waiting . . ;
call StabilizeRotation;
loop;
input tab autoLand;
tap tab setRotation;
branch setRotation SETROT;
goto SKIP;
lbl SETROT;
call ResetDesiredRotation;
print desiredAngle;
lbl SKIP;
AND autoLand landing runAutoLand;
branch runAutoLand AUTOLAND;
branch landing REBIND;
loop;
lbl REBIND;
L-RCS-L bind a;
L-RCS-R bind d;
LanderEngine bind w;
loop;
lbl AUTOLAND;
fetch VS directional descentV;
compare descentV 5 slowdown;
branch slowdown SLOW;
LanderEngine bind w;
goto STABILIZE;
lbl SLOW;
LanderEngine unbind;
LanderEngine startup;
lbl STABILIZE;
call StabilizeRotation;
